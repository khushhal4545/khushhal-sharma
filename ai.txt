# Import required library
from sklearn.ensemble import IsolationForest

# Sample operational data
data = [
    [20, 30],
    [25, 35],
    [22, 32],
    [95, 98]
]

# Create anomaly detection model
model = IsolationForest(
    contamination=0.1,
    random_state=42
)

# Train the model
model.fit(data)

# Detect anomalies
result = model.predict(data)

# Display result
for i in range(len(data)):
    if result[i] == -1:
        print(data[i], "Anomaly")
    else:
        print(data[i], "Normal")

from airflow import DAG
from airflow.operators.python import PythonOperator
from datetime import datetime


# Task 1: Extract data
def extract():
    print("Extracting operational data")


# Task 2: Transform data
def transform():
    print("Cleaning and transforming data")


# Task 3: Train model
def train():
    print("Training ML model")


# Create Airflow DAG
with DAG(
    dag_id="aiops_pipeline",
    start_date=datetime(2026, 1, 1),
    schedule=None,
    catchup=False
) as dag:

    extract_task = PythonOperator(
        task_id="extract_data",
        python_callable=extract
    )

    transform_task = PythonOperator(
        task_id="transform_data",
        python_callable=transform
    )

    train_task = PythonOperator(
        task_id="train_model",
        python_callable=train
    )

    # Define execution order
    extract_task >> transform_task >> train_task

from pyspark.sql import SparkSession

# Create Spark session
spark = SparkSession.builder \
    .appName("AIOps") \
    .getOrCreate()

# Sample server data
data = [
    ("Server1", 80),
    ("Server2", 95),
    ("Server3", 60)
]

# Create DataFrame
df = spark.createDataFrame(
    data,
    ["server", "cpu"]
)

# Display data
df.show()

# Select servers having CPU > 80
result = df.filter(df.cpu > 80)

# Display filtered result
result.show()

# Stop Spark

Track ML experiments using MLflow.

Code
import mlflow
import mlflow.sklearn

from sklearn.linear_model import LinearRegression

# Start MLflow experiment
with mlflow.start_run():

    # Training data
    X = [[1], [2], [3], [4]]
    y = [2, 4, 6, 8]

    # Create and train model
    model = LinearRegression()
    model.fit(X, y)

    # Calculate model score
    score = model.score(X, y)

    # Log metric
    mlflow.log_metric("score", score)

    # Save model information
    mlflow.sklearn.log_model(
        model,
        "linear_model"
    )

    print("Model tracked successfully")
